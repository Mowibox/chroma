{{- /* Local override of the Doks callout: same markup, but the template emits no indentation and no blank lines, so a callout survives being re-rendered as Markdown inside a Doks step or tab (otherwise the indented closing tags after a blank line become a code block). Usage unchanged: {{< callout context="note" title="Note" icon="outline/info-circle" >}}...{{< /callout >}} */ -}} {{- $css_class := "" -}} {{- if .IsNamedParams -}} {{- $css_class = .Get "context" | default "note" -}} {{- else -}} {{- $css_class = .Get 0 | default "note" -}} {{- end -}} {{- $icon := .Get "icon" -}} {{- $title := .Get "title" -}}
{{- with $icon }} {{ partial "inline-svg" (dict "src" $icon "class" "callout-icon me-2 mb-3") | replaceRE `\s*\n\s*` " " | safeHTML }} {{- end }}
{{- with $title }}
{{ . | $.Page.RenderString (dict "display" "block") | strings.TrimSpace | safeHTML }}
{{- end }}
{{ .Inner | $.Page.RenderString (dict "display" "block") | strings.TrimSpace | safeHTML }}
